using System.Text.Json;
using FlowNRW.Core.Transit;
var options = new TransitProviderOptions { MaxResults = 5 };
using var client = new HttpClient(new HttpClientHandler { AllowAutoRedirect = false }) { Timeout = Timeout.InfiniteTimeSpan };
var gateway = new TransitHttpGateway(client, options, new TransitDiagnostics(), new RetryPolicy());
var db = new DbRestProvider(gateway, new DbRestResponseMapper(options), options);
var efa = new EfaProvider(gateway, new EfaResponseMapper(options), options);
var orchestrator = new ProviderOrchestrator(db, efa, new NrwRegionClassifier(), new RealtimeConsolidator(), new MemoryTransitCache(new()), new());
var search = new StopSearchService(orchestrator, options);
var started = DateTimeOffset.UtcNow;
var locations = await search.SearchAsync("Berlin Hbf");
Console.WriteLine(JsonSerializer.Serialize(new { probe="search", started, locations.Source, locations.ErrorCode, locations.Warnings, locations.IsFallback, count=locations.Items.Count, first=locations.Items.FirstOrDefault() }));
var nearby = await search.NearbyAsync(new(51.505,7.096));
Console.WriteLine(JsonSerializer.Serialize(new { probe="nearby", nearby.RetrievedAt, nearby.Source, nearby.ErrorCode, nearby.Warnings, count=nearby.Items.Count, first=nearby.Items.FirstOrDefault() }));
var departures = await new DepartureService(orchestrator, options).DeparturesAsync(new() { Id="de:05513:5613", Name="Gelsenkirchen Hbf", Source="efa", Coordinate=new(51.505,7.096), Dhid="de:05513:5613" }, started);
Console.WriteLine(JsonSerializer.Serialize(new { probe="nrw-departures", departures.RetrievedAt, departures.Source, departures.ErrorCode, departures.Warnings, count=departures.Items.Count, first=departures.Items.FirstOrDefault() }));
var journeys = await new RoutingService(orchestrator, options).RouteAsync(new() { Name="Berlin Hbf", Stop=new() { Id="de:11000:900003200", Source="efa", Name="Berlin Hbf", Coordinate=new(52.5251,13.3694) } }, new() { Name="Hamburg Hbf", Coordinate=new(53.5527,10.0069) }, started);
Console.WriteLine(JsonSerializer.Serialize(new { probe="national-journeys", journeys.RetrievedAt, journeys.Source, journeys.ErrorCode, journeys.Warnings, journeys.IsFallback, count=journeys.Items.Count, first=journeys.Items.FirstOrDefault()?.Legs.Select(leg => new { from=leg.Departure.Identity.Stop.Name, to=leg.Arrival.Identity.Stop.Name, planned=leg.Departure.PlannedTime, arrival=leg.Arrival.PlannedTime, line=leg.Line?.Name, walking=leg.Walking is not null, geometry=leg.Geometry?.Coordinates.Count }) }));
return locations.HasData && nearby.HasData && departures.HasData && journeys.HasData ? 0 : 1;
